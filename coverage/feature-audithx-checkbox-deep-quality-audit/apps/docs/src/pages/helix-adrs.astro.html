      // Scroll-reveal animation using Intersection Observer
      // Respects prefers-reduced-motion
      (function () {
        var prefersReduced = window.matchMedia(
          '(prefers-reduced-motion: reduce)'
        ).matches;

        if (prefersReduced) {
          document.querySelectorAll('.scroll-reveal').forEach(function (el) {
            el.classList.add('is-visible');
          });
          document.querySelectorAll('.scroll-reveal-bar').forEach(function (el) {
            el.classList.add('is-visible');
          });
          return;
        }

        var observer = new IntersectionObserver(
          function (entries) {
            entries.forEach(function (entry) {
              if (entry.isIntersecting) {
                entry.target.classList.add('is-visible');
                observer.unobserve(entry.target);
              }
            });
          },
          {
            threshold: 0.1,
            rootMargin: '0px 0px -60px 0px',
          }
        );

        document.querySelectorAll('.scroll-reveal').forEach(function (el) {
          observer.observe(el);
        });

        // Performance bar fill animation on scroll
        var barObserver = new IntersectionObserver(
          function (entries) {
            entries.forEach(function (entry) {
              if (entry.isIntersecting) {
                entry.target.classList.add('is-visible');
                barObserver.unobserve(entry.target);
              }
            });
          },
          {
            threshold: 0.3,
            rootMargin: '0px 0px -40px 0px',
          }
        );

        document.querySelectorAll('.scroll-reveal-bar').forEach(function (el) {
          barObserver.observe(el);
        });

        // Cursor-following spotlight effect on cards
        document.querySelectorAll('.da-spotlight-card, .da-card').forEach(function (card) {
          card.addEventListener('mousemove', function (e) {
            var rect = card.getBoundingClientRect();
            var x = e.clientX - rect.left;
            var y = e.clientY - rect.top;
            card.style.setProperty('--mouse-x', x + 'px');
            card.style.setProperty('--mouse-y', y + 'px');
          });
        });

        // Tab ↔ hash slug mapping
        var tabHashMap = {
          'tab-slots-props': 'slots-vs-props',
          'tab-loading': 'component-loading',
          'tab-attributes': 'attribute-naming',
          'tab-lightdom': 'light-dom',
        };
        var hashTabMap = {};
        Object.keys(tabHashMap).forEach(function (k) { hashTabMap[tabHashMap[k]] = k; });

        // Tab switching
        var tabButtons = document.querySelectorAll('.da-nav-tab[role="tab"]');
        var tabPanels = document.querySelectorAll('.da-tab-content');

        function activateTab(tabId, updateHash) {
          // Update buttons
          tabButtons.forEach(function (b) {
            b.classList.remove('is-active');
            b.setAttribute('aria-selected', 'false');
          });
          var activeBtn = document.querySelector('.da-nav-tab[data-tab="' + tabId + '"]');
          if (activeBtn) {
            activeBtn.classList.add('is-active');
            activeBtn.setAttribute('aria-selected', 'true');
          }

          // Update panels
          tabPanels.forEach(function (panel) {
            panel.classList.remove('is-active');
          });
          var target = document.getElementById(tabId);
          if (target) {
            target.classList.add('is-active');

            // Update URL hash without scrolling
            if (updateHash && tabHashMap[tabId]) {
              history.replaceState(null, '', '#' + tabHashMap[tabId]);
            }

            // Scroll to top of content area
            window.scrollTo({ top: 0, behavior: 'smooth' });

            // Re-trigger scroll-reveal for newly visible panel
            setTimeout(function () {
              target.querySelectorAll('.scroll-reveal:not(.is-visible)').forEach(function (el) {
                observer.unobserve(el);
                observer.observe(el);
              });
              target.querySelectorAll('.scroll-reveal-bar:not(.is-visible)').forEach(function (el) {
                barObserver.unobserve(el);
                barObserver.observe(el);
              });
            }, 50);
          }
        }

        tabButtons.forEach(function (btn) {
          btn.addEventListener('click', function () {
            activateTab(btn.getAttribute('data-tab'), true);
          });
        });

        // Restore tab from URL hash on load
        var initialHash = window.location.hash.replace('#', '');
        if (initialHash && hashTabMap[initialHash]) {
          activateTab(hashTabMap[initialHash], false);
        }

        // Listen for hash changes (back/forward navigation)
        window.addEventListener('hashchange', function () {
          var hash = window.location.hash.replace('#', '');
          if (hash && hashTabMap[hash]) {
            activateTab(hashTabMap[hash], false);
          }
        });

        // Keyboard navigation for tabs
        tabButtons.forEach(function (btn, index) {
          btn.addEventListener('keydown', function (e) {
            var next = null;
            if (e.key === 'ArrowRight' || e.key === 'ArrowDown') {
              next = tabButtons[(index + 1) % tabButtons.length];
            } else if (e.key === 'ArrowLeft' || e.key === 'ArrowUp') {
              next = tabButtons[(index - 1 + tabButtons.length) % tabButtons.length];
            }
            if (next) {
              e.preventDefault();
              next.focus();
              next.click();
            }
          });
        });
      })();
